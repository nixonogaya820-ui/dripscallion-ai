/* DRIPSCALLION AI — VIDEO ATTACHMENT SUPPORT */ $("fileInput").setAttribute( "accept", "image/jpeg,image/png,image/webp,video/mp4,video/webm,video/quicktime,video/ogg,.mp4,.webm,.mov,.ogg,.pdf,.docx,.txt,.md,.csv,.xlsx,.xls,.xlsm,.xlsb,.et,.odt,.ods,.html,.htm,.xml,.numbers" ); function isVideoFile(file) { return [ "video/mp4", "video/webm", "video/quicktime", "video/ogg" ].includes((file.type || "").toLowerCase()) || [".mp4", ".webm", ".mov", ".ogg"].includes(extensionOf(file.name)); } function renderPendingFiles() { const wrap = $("attachmentPreview"); wrap.innerHTML = ""; pendingFiles.forEach((file, index) => { const chip = document.createElement("div"); chip.className = "attachment-chip"; if (isImageFile(file)) { const thumb = document.createElement("img"); thumb.src = URL.createObjectURL(file); thumb.alt = file.name; chip.appendChild(thumb); } else { const icon = document.createElement("div"); icon.className = "tool-icon"; icon.textContent = isVideoFile(file) ? "▶" : extensionOf(file.name).replace(".", "").slice(0, 4).toUpperCase() || "FILE"; chip.appendChild(icon); } const name = document.createElement("span"); name.textContent = file.name; name.title = file.name; const remove = document.createElement("button"); remove.type = "button"; remove.textContent = "×"; remove.title = "Remove attachment"; remove.onclick = () => { pendingFiles.splice(index, 1); renderPendingFiles(); }; chip.append(name, remove); wrap.appendChild(chip); }); } function addSelectedFiles(fileList) { const incoming = Array.from(fileList || []); const supported = [ ".pdf", ".docx", ".txt", ".md", ".csv", ".xlsx", ".xls", ".xlsm", ".xlsb", ".et", ".odt", ".ods", ".html", ".htm", ".xml", ".numbers" ]; for (const file of incoming) { if (pendingFiles.length >= 5) { alert("Attach up to five files at a time."); break; } const ext = extensionOf(file.name); const image = isImageFile(file); const video = isVideoFile(file); if (!image && !video && !supported.includes(ext)) { alert("Unsupported file: " + file.name); continue; } const max = image ? 5 * 1024 * 1024 : video ? 25 * 1024 * 1024 : 8 * 1024 * 1024; if (file.size > max) { alert( file.name + " is too large. Maximum size: " + (image ? "5 MB" : video ? "25 MB" : "8 MB") + "." ); continue; } pendingFiles.push(file); } renderPendingFiles(); } function extractVideoFrames(file) { return new Promise((resolve, reject) => { const video = document.createElement("video"); const url = URL.createObjectURL(file); video.preload = "metadata"; video.muted = true; video.playsInline = true; const cleanup = () => { URL.revokeObjectURL(url); video.removeAttribute("src"); video.load(); }; video.onerror = () => { cleanup(); reject(new Error("Could not read this video. Try MP4 or WebM.")); }; video.onloadedmetadata = async () => { try { if (!Number.isFinite(video.duration) || video.duration <= 0) { throw new Error("This video has invalid duration."); } if (video.duration > 120) { throw new Error("Video is too long. Maximum duration is 120 seconds."); } const width = video.videoWidth; const height = video.videoHeight; if (!width || !height) { throw new Error("Could not read the video dimensions."); } const scale = Math.min(1, 768 / width, 768 / height); const canvas = document.createElement("canvas"); canvas.width = Math.round(width * scale); canvas.height = Math.round(height * scale); const context = canvas.getContext("2d"); if (!context) throw new Error("Could not prepare video frames."); const timestamps = [ video.duration * 0.10, video.duration * 0.35, video.duration * 0.60, video.duration * 0.85 ]; const frames = []; for (const time of timestamps) { await new Promise((resolveSeek, rejectSeek) => { video.onseeked = resolveSeek; video.currentTime = Math.min(time, Math.max(0, video.duration - 0.1)); setTimeout(() => rejectSeek(new Error("Video frame loading timed out.")), 12000); }); context.drawImage(video, 0, 0, canvas.width, canvas.height); frames.push({ time, image: canvas.toDataURL("image/jpeg", 0.75) }); } cleanup(); resolve(frames); } catch (error) { cleanup(); reject(error); } }; video.src = url; }); } async function processAttachments(files, userQuestion) { const extraContext = []; const images = files.filter(isImageFile); const videos = files.filter(isVideoFile); const documents = files.filter(file => !isImageFile(file) && !isVideoFile(file)); for (const imageFile of images) { const imageData = await fileToDataURL(imageFile); const vision = await postJSON("/api/vision", { prompt: userQuestion || "Describe this image and explain its important details.", image: imageData }); extraContext.push( "IMAGE ANALYSIS (" + imageFile.name + "):\n" + (vision.response || "No image description was returned.") ); } for (const videoFile of videos) { const frames = await extractVideoFrames(videoFile); const analyses = []; for (let i = 0; i < frames.length; i++) { const result = await postJSON("/api/vision", { prompt: "Analyze this frame from a video. " + "The frame timestamp is " + frames[i].time.toFixed(1) + " seconds. Describe the visible scene, actions, objects and important details.", image: frames[i].image }); analyses.push( "Frame " + (i + 1) + " at " + frames[i].time.toFixed(1) + " seconds:\n" + (result.response || "No description returned.") ); } extraContext.push( "VIDEO ANALYSIS (" + videoFile.name + "):\n" + "The following four frames were sampled from the video:\n" + analyses.join("\n\n") ); } for (const file of documents) { let extractedText = ""; if (isTextFile(file)) { extractedText = await file.text(); } else { const fileData = await fileToDataURL(file); const converted = await postJSON("/api/document", { name: file.name, fileData }); extractedText = converted.text || ""; } if (!extractedText.trim()) { throw new Error("No readable text was extracted from " + file.name + "."); } extraContext.push( "DOCUMENT: " + file.name + "\n" + extractedText.slice(0, 18000) ); } return extraContext; }
